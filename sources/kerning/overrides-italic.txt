# Hand-set values for the pairs that matter most in prose and headings.
# Each value also moves the accented forms of the same letters by the
# same amount (tools/kerning.py).  Values come from a measured audit of
# the italic, checked in before/after proofs.  Format: LEFT RIGHT VALUE

T a -75

W o -70
W n -60
Y a -70
Y o -65
Y n -60
P a -50
P o -30
F a -25

# Sentence ends: the italic is much looser before . and , than the roman.
T period -55
T comma -55
V period -120
V comma -120
W period -100
W comma -100
P period -80
P comma -80
F period -40
F comma -40
r period -60
r comma -60
